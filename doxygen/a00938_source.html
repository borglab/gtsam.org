56template <typename KeyRange>
57inline size_t sumDims(const KeyDimMap& dims, const KeyRange& keys) {
58 size_t dim = 0;
59 for (Key key : keys) {
60 auto it = dims.find(key);
61 if (it != dims.end()) dim += it->second;
62 }
63 return dim;
64}
