39template<int Rows, int Cols>
40class OptionalJacobian {
41
42public:
43
46 typedef Eigen::Matrix<double, Rows, Cols> Jacobian;
47 typedef Eigen::Stride<Eigen::Dynamic, 1> Stride;
48 typedef Eigen::Map<Jacobian, 0, Stride> Map;
49
50private:
51
52 Map map_;
53
54 // Trick from http://eigen.tuxfamily.org/dox/group__TutorialMapClass.html
55 // uses "placement new" to make map_ usurp the memory of the fixed size matrix
56 void usurp(double* data, Eigen::Index outerStride) {
57 new (&map_) Map(data, Stride(outerStride, 1));
58 }
59
60 // Private and very dangerous constructor straight from memory
61 OptionalJacobian(double* data, Eigen::Index outerStride)
62 : map_(nullptr, Stride(1, 1)) {
63 if (data) usurp(data, outerStride);
64 }

69public:

73 }

79 map_(nullptr, Stride(1, 1)) {

96 OptionalJacobian(Eigen::MatrixXd& dynamic) :

99 usurp(dynamic.data(), dynamic.rows());
100 }

104 map_(nullptr, Stride(1, 1)) {
105 if (dynamic) {
106 dynamic->resize(Rows, Cols); // no malloc if correct size
107 usurp(dynamic->data(), dynamic->rows());
108 }
109 }

115 template<class MATRIX>
116 OptionalJacobian(Eigen::Ref<MATRIX> dynamic_ref) :
117 map_(nullptr, Stride(1, 1)) {
118 if (dynamic_ref.rows() == Rows && dynamic_ref.cols() == Cols &&
119 !dynamic_ref.IsRowMajor && dynamic_ref.innerStride() == 1) {
120 usurp(dynamic_ref.data(), dynamic_ref.outerStride());
121 } else {
122 throw std::invalid_argument(
123 std::string("OptionalJacobian called with wrong dimensions or "
124 "storage order.\n"
125 "Expected: ") +
126 "(" + std::to_string(Rows) + ", " + std::to_string(Cols) + ")");
127 }
128 }

131 OptionalJacobian(std::nullopt_t /*none*/) :
132 map_(nullptr, Stride(1, 1)) {
133 }
134
136 OptionalJacobian(const std::optional<std::reference_wrapper<Eigen::MatrixXd>> optional) :
137 map_(nullptr, Stride(1, 1)) {
138 if (optional) {
139 optional->get().resize(Rows, Cols);
140 usurp(optional->get().data(), optional->get().rows());
141 }
142 }
