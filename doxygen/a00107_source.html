195 Block range(DenseIndex startBlock, DenseIndex endBlock) {
196 assertInvariants();
197 DenseIndex actualStartBlock = startBlock + blockStart_;
198 DenseIndex actualEndBlock = endBlock + blockStart_;
199 if (startBlock != 0 || endBlock != 0) {
200 checkBlock(actualStartBlock);
201 assert(actualEndBlock < (DenseIndex)variableColOffsets_.size());
202 }
203 const DenseIndex startCol = variableColOffsets_[actualStartBlock];
204 const DenseIndex rangeCols = variableColOffsets_[actualEndBlock] - startCol;
205 return matrix_.block(rowStart_, startCol, this->rows(), rangeCols);
206 }

228 constBlock blockRows(DenseIndex block, DenseIndex beginRow,

232 const DenseIndex actualBlock = block + blockStart_;
233 checkBlock(actualBlock);
234 const DenseIndex startCol = variableColOffsets_[actualBlock];
235 const DenseIndex blockCols =

293 void checkBlock(DenseIndex block) const {
294 static_cast<void>(block); // Disable unused varibale warnings.
295 assert(matrix_.cols() == variableColOffsets_.back());
296 assert(block < (DenseIndex)variableColOffsets_.size() - 1);
297 assert(variableColOffsets_[block] < matrix_.cols() &&
298 variableColOffsets_[block + 1] <= matrix_.cols());

301 template <typename ITERATOR>
302 void fillOffsets(ITERATOR firstBlockDim, ITERATOR lastBlockDim,
303 bool appendOneDimension) {
304 variableColOffsets_[0] = 0;
305 DenseIndex j = 0;
306 for (ITERATOR dim = firstBlockDim; dim != lastBlockDim; ++dim, ++j)
307 variableColOffsets_[j + 1] = variableColOffsets_[j] + *dim;
308 if (appendOneDimension)
309 variableColOffsets_[j + 1] = variableColOffsets_[j] + 1;
310 }
