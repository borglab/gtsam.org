24 inline void HessianFactor::CheckAugmentedInformation(
25 size_t keyCount, const SymmetricBlockMatrix& augmentedInformation) {
26 const DenseIndex numBlocks = augmentedInformation.nBlocks();
27 if (numBlocks < 1 || static_cast<DenseIndex>(keyCount) != numBlocks - 1) {
28 throw std::invalid_argument(
29 "Error in HessianFactor constructor input. Number of provided keys plus\n"
30 "one for the information vector must equal the number of provided matrix blocks. ");
31 }
32
33 if (augmentedInformation.getDim(numBlocks - 1) != 1) {
34 throw std::invalid_argument(
35 "Error in HessianFactor constructor input. The last provided matrix block\n"
36 "must be the information vector, but the last provided block had more than one column.");
37 }
38 }
