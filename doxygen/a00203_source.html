60 static Weights DerivativeWeights(size_t N, double x, double a, double b) {
61 const double scale = coordinateScale(N, a, b);
62 return scale * calculateWeights(N, (x - a) * scale, 1);
63 }
64
65 private:
67 static double coordinateScale(size_t N, double a, double b) {
68 if (a == b) {
69 throw std::invalid_argument(
70 "CardinalSplineBasis interval must have nonzero length");
71 }
72 return (static_cast<double>(N > 0 ? N - 1 : 0) +
73 kernels::IrwinHallCDF2.getLength()) /
74 (b - a);
75 }
76
78 static Weights calculateWeights(size_t N, double x, size_t derivative) {
79 if (N == 0) return Weights(0);
80 if (N == 1) {
81 Weights weights(1);
82 weights(0) = derivative == 0 ? 1.0 : 0.0;
83 return weights;
84 }
85
86 const double kernelTime = x + kernels::IrwinHallCDF2.getBeginning();
87 Weights cumulative(N);
88 for (size_t index = 0; index < N; ++index) {
89 cumulative(static_cast<Eigen::Index>(index)) =
90 kernels::IrwinHallCDF2.evaluateDerivative(
91 derivative, kernelTime - static_cast<double>(index));
92 }
93
94 Weights weights(N);
95 weights(0) = derivative == 0 ? 1.0 - cumulative(1) : -cumulative(1);
96 for (size_t index = 1; index + 1 < N; ++index) {
97 weights(static_cast<Eigen::Index>(index)) =
98 cumulative(static_cast<Eigen::Index>(index)) -
99 cumulative(static_cast<Eigen::Index>(index + 1));
100 }
101 weights(static_cast<Eigen::Index>(N - 1)) =
102 cumulative(static_cast<Eigen::Index>(N - 1));
103 return weights;
104 }
105};
