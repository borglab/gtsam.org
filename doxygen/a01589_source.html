117 Vector evaluateError(const VectorN& v, OptionalMatrixType H) const override {
118 const double vnorm = v.norm();
119 if (H) {
120 if (vnorm > kMinNorm) {
121 *H = v.transpose() / vnorm;
122 } else {
123 // The norm is not differentiable at the origin. Return a zero
124 // Jacobian: the factor exerts no pull on a (near-)zero vector.
125 H->setZero(1, N);
126 }
127 }
128 return Vector1(vnorm - norm_);
129 }
