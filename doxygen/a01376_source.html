179 bool equals(const GncParams& other, double tol = 1e-9) const {
180 return baseOptimizerParams.equals(other.baseOptimizerParams)
181 && lossType == other.lossType && maxIterations == other.maxIterations
182 && std::fabs(lambdaStep - other.lambdaStep) <= tol
183 && scheduler == other.scheduler
