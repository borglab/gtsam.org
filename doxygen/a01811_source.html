142 void add(const MEASUREMENT& measured, const Key& poseKey,
143 const size_t& cameraId = 0) {
144 // store measurement and key
145 this->measured_.push_back(measured);
146 this->nonUniqueKeys_.push_back(poseKey);
147
148 // also store keys in the keys_ vector: these keys are assumed to be
149 // unique, so we avoid duplicates here
150 if (std::find(this->keys_.begin(), this->keys_.end(), poseKey) ==
151 this->keys_.end())
152 this->keys_.push_back(poseKey); // add only unique keys
153
154 // store id of the camera taking the measurement
155 cameraIds_.push_back(cameraId);
156 }

168 void add(const MEASUREMENTS& measurements, const KeyVector& poseKeys,
169 const FastVector<size_t>& cameraIds = FastVector<size_t>()) {
170 if (poseKeys.size() != measurements.size() ||
171 (poseKeys.size() != cameraIds.size() && cameraIds.size() != 0)) {
172 throw std::runtime_error(
173 "SmartProjectionRigFactor: "
174 "trying to add inconsistent inputs");
175 }
176 if (cameraIds.size() == 0 && cameraRig_->size() > 1) {
177 throw std::runtime_error(
178 "SmartProjectionRigFactor: "
179 "camera rig includes multiple camera "
180 "but add did not input cameraIds");
181 }
182 for (size_t i = 0; i < measurements.size(); i++) {
183 add(measurements[i], poseKeys[i],
184 cameraIds.size() == 0 ? 0 : cameraIds[i]);
185 }
186 }

231 typename Base::Cameras cameras(const Values& values) const override {
232 typename Base::Cameras cameras;
233 cameras.reserve(nonUniqueKeys_.size()); // preallocate
234 for (size_t i = 0; i < nonUniqueKeys_.size(); i++) {
235 const typename Base::Camera& camera_i = (*cameraRig_)[cameraIds_[i]];
236 const Pose3 world_P_sensor_i =
237 values.at<Pose3>(nonUniqueKeys_[i]) // = world_P_body
238 * camera_i.pose(); // = body_P_cam_i
239 cameras.emplace_back(world_P_sensor_i,
240 std::make_shared<typename CAMERA::CalibrationType>(
241 camera_i.calibration()));
242 }
243 return cameras;
244 }

266 void computeJacobiansWithTriangulatedPoint(typename Base::FBlocks& Fs,
267 Matrix& E, Vector& b,
268 const Cameras& cameras) const {
269 if (!this->result_) {
270 throw("computeJacobiansWithTriangulatedPoint");
271 } else { // valid result: compute jacobians
272 b = -cameras.reprojectionError(*this->result_, this->measured_, Fs, E);
273 for (size_t i = 0; i < Fs.size(); i++) {
274 const Pose3& body_P_sensor = (*cameraRig_)[cameraIds_[i]].pose();
275 const Pose3 world_P_body = cameras[i].pose() * body_P_sensor.inverse();
276 Eigen::Matrix<double, DimPose, DimPose> H;
277 world_P_body.compose(body_P_sensor, H);
278 Fs.at(i) = Fs.at(i) * H;
279 }
280 }

284 std::shared_ptr<RegularHessianFactor<DimPose> > createHessianFactor(
285 const Values& values, const double& lambda = 0.0,
286 bool diagonalDamping = false) const {
287 // we may have multiple observation sharing the same keys (e.g., 2 cameras
288 // measuring from the same body pose), hence the number of unique keys may
289 // be smaller than nrMeasurements
290 size_t nrUniqueKeys =
291 this->keys_
292 .size(); // note: by construction, keys_ only contains unique keys
293
294 Cameras cameras = this->cameras(values);
295
296 // Create structures for Hessian Factors
297 std::vector<size_t> js;
298 std::vector<Matrix> Gs(nrUniqueKeys * (nrUniqueKeys + 1) / 2);
299 std::vector<Vector> gs(nrUniqueKeys);
300
301 if (this->measured_.size() != cameras.size()) // 1 observation per camera
302 throw std::runtime_error(
303 "SmartProjectionRigFactor: "
304 "measured_.size() inconsistent with input");
305
306 // triangulate 3D point at given linearization point
307 this->triangulateSafe(cameras);
308
309 if (!this->result_) { // failed: return "empty/zero" Hessian
310 if (this->params_.degeneracyMode == ZERO_ON_DEGENERACY) {
311 for (Matrix& m : Gs) m = Matrix::Zero(DimPose, DimPose);
312 for (Vector& v : gs) v = Vector::Zero(DimPose);
313 return std::make_shared<RegularHessianFactor<DimPose> >(this->keys_, Gs,
314 gs, 0.0);
315 } else {
316 throw std::runtime_error(
317 "SmartProjectionRigFactor: "
318 "only supported degeneracy mode is ZERO_ON_DEGENERACY");
319 }
320 }
321
322 // compute Jacobian given triangulated 3D Point
323 typename Base::FBlocks Fs;
324 Matrix E;
325 Vector b;
326 this->computeJacobiansWithTriangulatedPoint(Fs, E, b, cameras);
327
328 // Whiten using noise model
329 this->noiseModel_->WhitenSystem(E, b);
330 for (size_t i = 0; i < Fs.size(); i++) {
331 Fs[i] = this->noiseModel_->Whiten(Fs[i]);
332 }
333
334 const Matrix3 P = Base::Cameras::PointCov(E, lambda, diagonalDamping);
335
336 // Build augmented Hessian (with last row/column being the information
337 // vector) Note: we need to get the augumented hessian wrt the unique keys
338 // in key_
339 SymmetricBlockMatrix augmentedHessianUniqueKeys =
340 Base::Cameras::template SchurComplementAndRearrangeBlocks<3, 6, 6>(
341 Fs, E, P, b, nonUniqueKeys_, this->keys_);
342
343 return std::make_shared<RegularHessianFactor<DimPose> >(
344 this->keys_, augmentedHessianUniqueKeys);
345 }

354 std::shared_ptr<GaussianFactor> linearizeDamped(
355 const Values& values, const double& lambda = 0.0) const {
356 // depending on flag set on construction we may linearize to different
357 // linear factors
358 switch (this->params_.linearizationMode) {
359 case HESSIAN:
360 return this->createHessianFactor(values, lambda);
361 default:
362 throw std::runtime_error(
363 "SmartProjectionRigFactor: unknown linearization mode");
364 }
365 }
