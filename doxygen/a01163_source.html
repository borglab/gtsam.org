203 // φ: autonomous flow where velocity acts on position for
204 // dt (R, p, v) -> p += v·dt.
205 struct AutonomousFlow {
206 double dt;
207
208 // Differential at identity: Φ = I with ∂p/∂v = dt·I.
209 Jacobian dIdentity() const {
210 Jacobian Phi = I_9x9;
211 Phi.template block<3, 3>(3, 6) = I_3x3 * dt;
212 return Phi;
213 }
214
215 // Apply φ(x) by p += v·dt
216 NavState operator()(const NavState& X) const {
217 return {X.attitude(), X.position() + X.velocity() * dt, X.velocity()};
218 }
219 };
