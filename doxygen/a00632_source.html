39template <int N>
40SO<N> SO<N>::ChartAtOrigin::Retract(const TangentVector& xi) {
41 const Matrix X = Hat(xi / 2.0);
42 size_t n = AmbientDim(xi.size());
43 const auto I = Eigen::MatrixXd::Identity(n, n);
44 // https://pdfs.semanticscholar.org/6165/0347b2ccac34b5f423081d1ce4dbc4d09475.pdf
45 return SO((I + X) * (I - X).inverse());
46}

48template <int N>
49typename SO<N>::TangentVector SO<N>::ChartAtOrigin::Local(const SO& R) {
50 const size_t n = R.rows();
51 const auto I = Eigen::MatrixXd::Identity(n, n);
52 const Matrix X = (I - R.matrix_) * (I + R.matrix_).inverse();
53 return -2 * Vee(X);
54}
