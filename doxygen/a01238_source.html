190 Matrix dIdentity() const {
191 Matrix Phi = Matrix::Identity(9 + 3 * static_cast<int>(numFeet),
192 9 + 3 * static_cast<int>(numFeet));
193 Phi.block(3, 6, 3, 3) = I_3x3 * dt;
194 return Phi;
195 }
196
198 ExtendedPose3d operator()(const ExtendedPose3d& state) const {
199 Matrix blocks = state.xMatrix();
200 blocks.col(0) += blocks.col(1) * dt;
201 return ExtendedPose3d(state.rotation(), blocks);
202 }
