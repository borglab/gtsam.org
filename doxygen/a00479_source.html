292 template<class CalibrationB>
293 double range(const PinholeCamera<CalibrationB>& camera,
294 OptionalJacobian<1, dimension> Dcamera = {},
295 OptionalJacobian<1, 6 + CalibrationB::dimension> Dother = {}) const {
296 Matrix16 Dcamera_ = Matrix16::Zero(), Dother_ = Matrix16::Zero();
297 double result = this->pose().range(camera.pose(), Dcamera ? &Dcamera_ : 0,
298 Dother ? &Dother_ : 0);
299 if (Dcamera) {
300 *Dcamera << Dcamera_, Eigen::Matrix<double, 1, DimK>::Zero();
301 }
302 if (Dother) {
303 Dother->setZero();
304 Dother->template block<1, 6>(0, 0) = Dother_;
305 }
306 return result;
307 }
