243 ErrorVector evaluateError(const A1& a1, const A2& a2, const T& bias,
244 OptionalMatrixType H1 = OptionalNone,
245 OptionalMatrixType H2 = OptionalNone,
246 OptionalMatrixType H3 = OptionalNone) const override {
247 Matrix HposeCompose;
248 const A1 nav_T_sensor =
249 a1.compose(body_T_sensor_, H1 ? &HposeCompose : nullptr);
250 const T predictedRange = Range<A1, A2>()(nav_T_sensor, a2, H1, H2);
251 if (H1) *H1 *= HposeCompose;
252 if (H3) {
253 if constexpr (std::is_same_v<T, double>) {
254 *H3 = Matrix1::Identity();
255 } else {
256 const int dimension = traits<T>::GetDimension(measured_);
257 *H3 = Matrix::Identity(dimension, dimension);
258 }
259 }
260 return -traits<T>::Local(predictedRange + bias, measured_);
261 }
