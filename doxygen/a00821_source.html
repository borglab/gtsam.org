308 template<class ARCHIVE>
309 void save(ARCHIVE& ar, const unsigned int version) const {
310 if (version == 0) {
311 ar & BOOST_SERIALIZATION_NVP(nodes_);
312 ar & BOOST_SERIALIZATION_NVP(roots_);
313 return;
314 }
315
316 std::vector<sharedClique> cliques, detached;
317 std::unordered_map<const Clique*, size_t> indices;
318 for (const sharedClique& root : roots_) {
319 if (!indices.emplace(root.get(), cliques.size()).second)
320 throw std::invalid_argument("BayesTree contains duplicate cliques");
321 cliques.push_back(root);
322 }
323
324 std::vector<std::vector<size_t>> children;
325 for (size_t i = 0; i < cliques.size(); ++i) {
326 sharedClique copy = std::make_shared<Clique>(*cliques[i]);
327 copy->parent_.reset();
328 copy->children.clear();
329 copy->is_root = true;
330 detached.push_back(copy);
331 children.emplace_back();
332 for (const sharedClique& child : cliques[i]->children) {
333 if (!indices.emplace(child.get(), cliques.size()).second)
334 throw std::invalid_argument("BayesTree contains duplicate cliques");
335 children.back().push_back(cliques.size());
336 cliques.push_back(child);
337 }
338 }
339
340 const size_t rootCount = roots_.size();
341 ar & BOOST_SERIALIZATION_NVP(rootCount);
342 ar & boost::serialization::make_nvp("cliques", detached);
343 ar & BOOST_SERIALIZATION_NVP(children);
344 }
345
347 template<class ARCHIVE>
348 void load(ARCHIVE& ar, const unsigned int version) {
349 if (version == 0) {
350 ar & BOOST_SERIALIZATION_NVP(nodes_);
351 ar & BOOST_SERIALIZATION_NVP(roots_);
352 return;
353 }
354
355 size_t rootCount;
356 std::vector<sharedClique> cliques;
357 std::vector<std::vector<size_t>> children;
358 ar & BOOST_SERIALIZATION_NVP(rootCount);
359 ar & BOOST_SERIALIZATION_NVP(cliques);
360 ar & BOOST_SERIALIZATION_NVP(children);
361 if (rootCount > cliques.size() || children.size() != cliques.size())
362 throw std::invalid_argument("Invalid serialized BayesTree structure");
363
364 nodes_.clear();
365 roots_.assign(cliques.begin(), cliques.begin() + rootCount);
366 for (size_t i = 0; i < cliques.size(); ++i) {
367 sharedClique& clique = cliques[i];
368 clique->parent_.reset();
369 clique->children.clear();
370 clique->is_root = i < rootCount;
371 }
372 for (size_t i = 0; i < cliques.size(); ++i) {
373 sharedClique& clique = cliques[i];
374 for (size_t childIndex : children[i]) {
375 if (childIndex >= cliques.size())
376 throw std::invalid_argument("Invalid serialized BayesTree child");
377 sharedClique& child = cliques[childIndex];
378 if (!child->parent_.expired())
379 throw std::invalid_argument("Serialized BayesTree child has multiple parents");
380 clique->children.push_back(child);
381 child->parent_ = clique;
382 child->is_root = false;
383 }
384 }
385
386 for (size_t i = 0; i < cliques.size(); ++i) {
387 if ((i < rootCount) != cliques[i]->isRoot())
388 throw std::invalid_argument("Invalid serialized BayesTree roots");
389 for (Key key : cliques[i]->conditional()->frontals()) {
390 if (!nodes_.insert({key, cliques[i]}).second)
391 throw std::invalid_argument("Serialized BayesTree contains duplicate keys");
392 }
393 }
394 }
