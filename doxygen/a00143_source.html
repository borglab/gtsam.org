87template<class Class, int N>
88struct GetDimensionImpl {
89 // Get dimension at compile-time for fixed-size manifolds, and at
90 // run-time for dynamic-size manifolds.
91 static size_t GetDimension(const Class& m) {
92 if constexpr (N == Eigen::Dynamic) {
93 return m.dim();
94 } else {
95 return static_cast<size_t>(N);

103template<class Class>
104struct ManifoldTraits: GetDimensionImpl<Class, Class::dimension> {
105
106 // Check that Class has the necessary machinery
107 GTSAM_CONCEPT_ASSERT(HasManifoldPrereqs<Class>);
108
109 // Dimension of the manifold
110 inline constexpr static auto dimension = Class::dimension;
111
112 // Typedefs required by all manifold types.
113 typedef Class ManifoldType;
114 typedef manifold_tag structure_category;
115 typedef Eigen::Matrix<double, dimension, 1> TangentVector;
116 typedef OptionalJacobian<dimension, dimension> ChartJacobian;
117
118 // Local coordinates
119 static TangentVector Local(const Class& origin, const Class& other) {
120 return origin.localCoordinates(other);
121 }
122
123 template <class C = Class,
124 class = LocalCoordinatesWithJacobians<C, ChartJacobian>>
125 static TangentVector Local(const Class& origin, const Class& other,
126 ChartJacobian H1, ChartJacobian H2 = {}) {
127 return origin.localCoordinates(other, H1, H2);
128 }
129
130 // Retraction back to manifold
131 static Class Retract(const Class& origin, const TangentVector& v) {
132 return origin.retract(v);
133 }
134
135 template <class C = Class,
136 class = RetractWithJacobians<C, TangentVector, ChartJacobian>>
137 static Class Retract(const Class& origin, const TangentVector& v,
138 ChartJacobian H1, ChartJacobian H2 = {}) {
139 return origin.retract(v, H1, H2);
140 }
141};

161template<typename T>
162GTSAM_CONCEPT_REQUIRES(IsTestable<T>, bool) //
163check_manifold_invariants(const T& a, const T& b, double tol=1e-9) {
164 typename traits<T>::TangentVector v0 = traits<T>::Local(a,a);
165 typename traits<T>::TangentVector v = traits<T>::Local(a,b);
166 T c = traits<T>::Retract(a,v);
167 return v0.norm() < tol && traits<T>::Equals(b,c,tol);
168}

171template<typename T>
172class IsManifold {
173
174public:
175
176 using structure_category_tag = typename traits<T>::structure_category;
177 static inline constexpr int dim = traits<T>::dimension;
178 using ManifoldType = typename traits<T>::ManifoldType;
179 using TangentVector = typename traits<T>::TangentVector;
180 // Concept marker: allows checking IsManifold<T>::value in templates
181 static constexpr bool value =
182 std::is_base_of_v<manifold_tag, structure_category_tag>;
183
184 GTSAM_CONCEPT_USAGE(IsManifold) {
185 static_assert(
186 value,
187 "This type's structure_category trait does not assert it as a manifold (or derived)");
188 if constexpr (dim != Eigen::Dynamic) {
189 static_assert(TangentVector::SizeAtCompileTime == dim);
190 }
191
192 // make sure Chart methods are defined
193 v = traits<T>::Local(p, q);
194 q = traits<T>::Retract(p, v);
195 }
196
197private:
198
199 TangentVector v;
200 ManifoldType p, q;
201};
